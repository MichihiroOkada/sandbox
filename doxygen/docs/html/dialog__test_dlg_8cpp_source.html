77BOOL CdialogtestDlg::OnInitDialog()
78{
79 CDialogEx::OnInitDialog();
80
81 // "バージョン情報..." メニューをシステム メニューに追加します。
82
83 // IDM_ABOUTBOX は、システム コマンドの範囲内になければなりません。
84 ASSERT((IDM_ABOUTBOX & 0xFFF0) == IDM_ABOUTBOX);
85 ASSERT(IDM_ABOUTBOX < 0xF000);
86
88 CMenu* pSysMenu = GetSystemMenu(FALSE);
89 if (pSysMenu != nullptr)
90 {
91 BOOL bNameValid;
92 CString strAboutMenu;
94 bNameValid = strAboutMenu.LoadString(IDS_ABOUTBOX);
95 ASSERT(bNameValid);
96 if (!strAboutMenu.IsEmpty())
97 {
99 pSysMenu->AppendMenu(MF_SEPARATOR);
100 pSysMenu->AppendMenu(MF_STRING, IDM_ABOUTBOX, strAboutMenu);
101 }
102 }
103
104 // このダイアログのアイコンを設定します。アプリケーションのメイン ウィンドウがダイアログでない場合、
105 // Framework は、この設定を自動的に行います。
106 SetIcon(m_hIcon, TRUE); // 大きいアイコンの設定
107 SetIcon(m_hIcon, FALSE); // 小さいアイコンの設定
108
109 // TODO: 初期化をここに追加します。
110
111 return TRUE; // フォーカスをコントロールに設定した場合を除き、TRUE を返します。

133void CdialogtestDlg::OnPaint()
134{
135 if (IsIconic())
136 {
137 CPaintDC dc(this); // 描画のデバイス コンテキスト
138
139 SendMessage(WM_ICONERASEBKGND, reinterpret_cast<WPARAM>(dc.GetSafeHdc()), 0);
140
141 // クライアントの四角形領域内の中央
142 int cxIcon = GetSystemMetrics(SM_CXICON);
143 int cyIcon = GetSystemMetrics(SM_CYICON);
144 CRect rect;
145 GetClientRect(&rect);
146 int x = (rect.Width() - cxIcon + 1) / 2;
147 int y = (rect.Height() - cyIcon + 1) / 2;
148
149 // アイコンの描画
150 dc.DrawIcon(x, y, m_hIcon);
151 }
152 else
153 {
154 CDialogEx::OnPaint();
155 }
156}
